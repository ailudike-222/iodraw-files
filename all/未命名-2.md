import matplotlib.pyplot as plt
import matplotlib.patches as patches
from matplotlib.patches import FancyArrowPatch

plt.rcParams['font.sans-serif'] = ['WenQuanYi Zen Hei']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(18, 14))
ax.set_xlim(0, 14)
ax.set_ylim(0, 14)
ax.axis('off')

system = patches.Rectangle((2, 2), 10, 10, linewidth=4, edgecolor='#1A237E', facecolor='none')
ax.add_patch(system)
ax.text(7, 12.5, '图书管理系统', ha='center', va='center', fontsize=24, fontweight='bold', color='#1A237E')

actors = {
    '读者': (1, 9.5),
    '图书管理员': (1, 7),
    '系统管理员': (1, 4.5)
}

colors = ['#0D47A1', '#E65100', '#00695C']
for i, (name, (x, y)) in enumerate(actors.items()):
    color = colors[i]
    circle = patches.Circle((x, y+0.5), 0.25, color=color)
    ax.add_patch(circle)
    ax.plot([x, x], [y, y+0.25], color=color, linewidth=4)
    ax.plot([x-0.25, x+0.25], [y+0.2, y+0.2], color=color, linewidth=4)
    ax.plot([x, x-0.25], [y, y-0.35], color=color, linewidth=4)
    ax.plot([x, x+0.25], [y, y-0.35], color=color, linewidth=4)
    ax.text(x, y-0.7, name, ha='center', va='center', fontsize=16, fontweight='bold', color=color)

use_cases = {
    '查询图书': (3.5, 10),
    '借阅图书': (3.5, 9),
    '归还图书': (3.5, 8),
    '续借图书': (3.5, 7),
    '图书入库': (7, 9.5),
    '图书删除': (7, 8.5),
    '借阅处理': (7, 7.5),
    '归还处理': (7, 6.5),
    '读者信息管理': (7, 5.5),
    '账号管理': (10.5, 9),
    '权限设置': (10.5, 8),
    '系统维护': (10.5, 7),
    '数据备份': (10.5, 6)
}

for name, (x, y) in use_cases.items():
    ellipse = patches.Ellipse((x, y), 2.2, 1, edgecolor='#263238', facecolor='#FAFAFA', linewidth=2.5)
    ax.add_patch(ellipse)
    ax.text(x, y, name, ha='center', va='center', fontsize=14, fontweight='bold')

reader_cases = ['查询图书', '借阅图书', '归还图书', '续借图书']
for uc in reader_cases:
    x1, y1 = actors['读者']
    x2, y2 = use_cases[uc]
    ax.plot([x1+0.5, x2-1.1], [y1, y2], color='#0D47A1', linewidth=2.5)

lib_cases = ['图书入库', '图书删除', '借阅处理', '归还处理', '读者信息管理']
for uc in lib_cases:
    x1, y1 = actors['图书管理员']
    x2, y2 = use_cases[uc]
    ax.plot([x1+0.5, x2-1.1], [y1, y2], color='#E65100', linewidth=2.5)

admin_cases = ['账号管理', '权限设置', '系统维护', '数据备份']
for uc in admin_cases:
    x1, y1 = actors['系统管理员']
    x2, y2 = use_cases[uc]
    ax.plot([x1+0.5, x2-1.1], [y1, y2], color='#00695C', linewidth=2.5)

arrow1 = FancyArrowPatch((4, 9), (6, 7.5), arrowstyle='->', mutation_scale=25, linestyle='dashed', color='#B71C1C', linewidth=2.5)
ax.add_patch(arrow1)
ax.text(4.8, 8.3, '<<include>>', ha='center', va='center', fontsize=13, fontweight='bold', color='#B71C1C')

arrow2 = FancyArrowPatch((4, 8), (6, 6.5), arrowstyle='->', mutation_scale=25, linestyle='dashed', color='#B71C1C', linewidth=2.5)
ax.add_patch(arrow2)
ax.text(4.8, 7.3, '<<include>>', ha='center', va='center', fontsize=13, fontweight='bold', color='#B71C1C')

plt.tight_layout()
plt.savefig('/mnt/图书管理系统用例图_600DPI.png', dpi=600, bbox_inches='tight', facecolor='white')
plt.close()
print("600DPI超清用例图生成完成")
